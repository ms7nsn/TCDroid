package com.tcdroid.core;

import java.io.File;
import java.io.IOException;
import java.io.RandomAccessFile;

/** Implémentation JVM (tests) ; sur Android on utilise ChannelStorage. */
public final class FileStorage implements Storage {
    private final RandomAccessFile raf;
    public FileStorage(File f, boolean writable) throws IOException {
        raf = new RandomAccessFile(f, writable ? "rw" : "r");
    }
    @Override public long length() throws IOException { return raf.length(); }
    @Override public void read(long o, byte[] b, int off, int len) throws IOException {
        raf.seek(o); raf.readFully(b, off, len);
    }
    @Override public void write(long o, byte[] b, int off, int len) throws IOException {
        raf.seek(o); raf.write(b, off, len);
    }
    @Override public void flush() throws IOException { raf.getFD().sync(); }
    @Override public void close() throws IOException { raf.close(); }
}
